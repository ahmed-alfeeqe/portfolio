YoY Revenue Growth % = 
VAR CurrentRevenue = [Total Revenue]
VAR PriorYearRevenue = 
    CALCULATE(
        [Total Revenue],
        SAMEPERIODLASTYEAR('DimCalendar'[Date])
    )
RETURN
    DIVIDE(
        CurrentRevenue - PriorYearRevenue,
        PriorYearRevenue,
        0
    )

DIVIDE(
    [Total Revenue] - [Total Product Cost],
    [Total Revenue],
    0
